// rclicker — phone controller.
//
// The QR code opens https://<relay>/#k=<key>. The key stays in the URL fragment, which
// browsers never send to the server. From it we derive (HMAC-SHA-256):
//   room id      which relay room to join (not secret)
//   phone token  proves to the relay we scanned this QR code (relay only stores its hash)
//   AES key      end-to-end encryption with the PC; the relay cannot read or forge commands
// Only whitelisted semantic commands ("presentation.next", ...) are ever sent.
(function () {
  'use strict';

  var TAP_LOCK_MS = 300;        // Ignore a second tap on the same button within this window.
  var ACK_TIMEOUT_MS = 4000;    // No reply this long after a command => connection is probably dead.
  var PING_EVERY_MS = 15000;    // Keep-alive while the page is visible.
  var PONG_TIMEOUT_MS = 5000;
  var MAX_BACKOFF_MS = 8000;
  var MESSAGE_MS = 4000;

  var PING = '{"t":"ping"}';    // Must match the relay's auto-response text exactly.

  var CLOSE_SESSION_ENDED = 4401;
  var CLOSE_REPLACED = 4408;
  var CLOSE_BUSY = 4409;
  var CLOSE_RECEIVER_CLOSED = 4410;

  var encoder = new TextEncoder();
  var decoder = new TextDecoder();
  var clientId = getClientId();

  var statusEl = document.getElementById('status');
  var statusText = document.getElementById('status-text');
  var messageEl = document.getElementById('message');
  var awakeEl = document.getElementById('awake');
  var overlay = document.getElementById('overlay');
  var overlayTitle = document.getElementById('overlay-title');
  var overlayText = document.getElementById('overlay-text');
  var overlayRetry = document.getElementById('overlay-retry');
  var buttons = Array.prototype.slice.call(document.querySelectorAll('[data-command]'));

  var keys = null;            // { room, phoneToken, aesKey }
  var ws = null;
  var state = 'connecting';
  var stopped = false;        // True once the session is over; no more reconnects.
  var everConnected = false;
  var hostOnline = false;
  var nonce = null;           // Per-connection challenge from the PC; proves our commands are fresh.
  var attempt = 0;
  var nextId = 1;
  var reconnectTimer = null;
  var pingTimer = null;
  var pongTimer = null;
  var ackTimers = {};
  var lastTap = {};
  var messageTimer = null;
  var wakeLock = null;

  // ---------- crypto ----------

  function fromBase64Url(text) {
    var b64 = text.replace(/-/g, '+').replace(/_/g, '/');
    while (b64.length % 4) { b64 += '='; }
    var binary = atob(b64);
    var bytes = new Uint8Array(binary.length);
    for (var i = 0; i < binary.length; i++) { bytes[i] = binary.charCodeAt(i); }
    return bytes;
  }

  function toBase64Url(bytes) {
    var binary = '';
    for (var i = 0; i < bytes.length; i++) { binary += String.fromCharCode(bytes[i]); }
    return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
  }

  function deriveKeys(keyBytes) {
    var subtle = window.crypto.subtle;
    return subtle.importKey('raw', keyBytes, { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']).then(function (hmac) {
      function mac(label) {
        return subtle.sign('HMAC', hmac, encoder.encode(label)).then(function (b) { return new Uint8Array(b); });
      }
      return Promise.all([mac('rclicker/v1/room'), mac('rclicker/v1/phone'), mac('rclicker/v1/enc')]);
    }).then(function (macs) {
      return subtle.importKey('raw', macs[2], { name: 'AES-GCM' }, false, ['encrypt', 'decrypt']).then(function (aesKey) {
        return { room: toBase64Url(macs[0].slice(0, 16)), phoneToken: toBase64Url(macs[1]), aesKey: aesKey };
      });
    });
  }

  function encrypt(payload) {
    var iv = window.crypto.getRandomValues(new Uint8Array(12));
    return window.crypto.subtle.encrypt(
      { name: 'AES-GCM', iv: iv, additionalData: encoder.encode('rclicker/v1/p2h/' + keys.room) },
      keys.aesKey,
      encoder.encode(JSON.stringify(payload))
    ).then(function (ct) {
      return JSON.stringify({ t: 'msg', iv: toBase64Url(iv), ct: toBase64Url(new Uint8Array(ct)) });
    });
  }

  function decrypt(iv, ct) {
    return window.crypto.subtle.decrypt(
      { name: 'AES-GCM', iv: fromBase64Url(iv), additionalData: encoder.encode('rclicker/v1/h2p/' + keys.room) },
      keys.aesKey,
      fromBase64Url(ct)
    ).then(function (plain) { return JSON.parse(decoder.decode(plain)); });
  }

  // ---------- connection ----------

  function connect() {
    clearTimeout(reconnectTimer);
    reconnectTimer = null;
    if (stopped) { return; }
    if (ws && (ws.readyState === WebSocket.CONNECTING || ws.readyState === WebSocket.OPEN)) { return; }

    setState(everConnected ? 'reconnecting' : 'connecting');
    var scheme = window.location.protocol === 'https:' ? 'wss://' : 'ws://';
    var url = scheme + window.location.host + '/ws/phone?room=' + encodeURIComponent(keys.room);

    var socket;
    try {
      socket = new WebSocket(url);
    } catch (e) {
      scheduleReconnect();
      return;
    }

    ws = socket;
    nonce = null;

    socket.onopen = function () {
      if (ws !== socket) { return; }
      attempt = 0;
      socket.send(JSON.stringify({ t: 'auth', token: keys.phoneToken, client: clientId }));
      startHeartbeat();
    };

    socket.onmessage = function (event) {
      if (ws !== socket) { return; }
      handleRelayMessage(event.data);
    };

    socket.onerror = function () { /* onclose follows */ };

    socket.onclose = function (event) {
      if (ws !== socket) { return; }
      ws = null;
      nonce = null;
      stopHeartbeat();
      clearAckTimers();
      handleClose(event.code);
    };
  }

  function handleClose(code) {
    switch (code) {
      case CLOSE_SESSION_ENDED:
        return endSession('Session ended',
          'This QR code is no longer valid. Scan the QR code shown in rclicker on the computer.');
      case CLOSE_RECEIVER_CLOSED:
        return endSession('rclicker closed',
          'rclicker was closed on the computer. Start it again and scan the new QR code.');
      case CLOSE_BUSY:
        return pause('Another phone is in control',
          'Only one phone can control the presentation at a time. Ask the presenter to close the remote on their phone, then try again.');
      case CLOSE_REPLACED:
        return pause('Opened somewhere else',
          'The remote was opened in another tab or window on this device. Use that one, or tap below to use this one instead.');
    }
    scheduleReconnect();
  }

  function scheduleReconnect() {
    if (stopped || reconnectTimer) { return; }
    setState(everConnected ? 'reconnecting' : 'connecting');
    var delay = Math.min(MAX_BACKOFF_MS, 400 * Math.pow(2, attempt));
    attempt += 1;
    if (attempt > 5) {
      showMessage("Can't reach the internet. Check the phone's mobile data or Wi-Fi.", true);
    }
    reconnectTimer = setTimeout(function () {
      reconnectTimer = null;
      connect();
    }, delay);
  }

  // Throw away a socket that is probably dead (no pong / no ack) and reconnect now.
  function dropAndReconnect() {
    var socket = ws;
    ws = null;
    nonce = null;
    stopHeartbeat();
    clearAckTimers();
    if (socket) {
      socket.onopen = socket.onmessage = socket.onclose = socket.onerror = null;
      try { socket.close(); } catch (e) { /* ignore */ }
    }
    attempt = 0;
    if (!stopped) { connect(); }
  }

  function startHeartbeat() {
    stopHeartbeat();
    pingTimer = setInterval(function () {
      if (document.visibilityState === 'visible') { ping(); }
    }, PING_EVERY_MS);
  }

  function stopHeartbeat() {
    clearInterval(pingTimer);
    clearTimeout(pongTimer);
    pingTimer = null;
    pongTimer = null;
  }

  function ping() {
    if (!ws || ws.readyState !== WebSocket.OPEN || pongTimer) { return; }
    try {
      ws.send(PING);
    } catch (e) {
      dropAndReconnect();
      return;
    }
    pongTimer = setTimeout(function () {
      pongTimer = null;
      dropAndReconnect();
    }, PONG_TIMEOUT_MS);
  }

  // ---------- messages ----------

  function handleRelayMessage(data) {
    var msg;
    try {
      msg = JSON.parse(data);
    } catch (e) {
      return;
    }
    if (!msg || typeof msg.t !== 'string') { return; }

    switch (msg.t) {
      case 'pong':
        clearTimeout(pongTimer);
        pongTimer = null;
        break;
      case 'host':
        hostOnline = !!msg.online;
        if (!hostOnline) {
          nonce = null;
          clearAckTimers();
        }
        setState(hostOnline && nonce ? 'connected' : 'waiting');
        break;
      case 'msg':
        if (typeof msg.iv !== 'string' || typeof msg.ct !== 'string') { return; }
        decrypt(msg.iv, msg.ct).then(handleSecureMessage, function () {
          // Not from our PC (or corrupted): ignore.
        });
        break;
    }
  }

  function handleSecureMessage(msg) {
    if (!msg || typeof msg.type !== 'string') { return; }

    switch (msg.type) {
      case 'hello':
        if (typeof msg.nonce !== 'string') { return; }
        nonce = msg.nonce;
        hostOnline = true;
        everConnected = true;
        clearMessage();
        setState('connected');
        break;
      case 'ack':
        settle(msg.id);
        if (msg.status === 'ok') {
          flash(msg.command, true);
        } else if (msg.status === 'rate_limited') {
          // A double tap was ignored on purpose; no need to alarm anyone.
        } else {
          flash(msg.command, false);
          showMessage(msg.message || 'The computer could not do that.', true);
        }
        break;
      case 'error':
        settle(msg.id);
        showMessage('The computer did not understand that. Reload the page if this keeps happening.', true);
        break;
    }
  }

  function settle(id) {
    if (typeof id === 'number' && ackTimers[id]) {
      clearTimeout(ackTimers[id]);
      delete ackTimers[id];
    }
  }

  function clearAckTimers() {
    Object.keys(ackTimers).forEach(function (id) { clearTimeout(ackTimers[id]); });
    ackTimers = {};
  }

  // ---------- commands ----------

  function sendCommand(command) {
    var now = Date.now();
    if (now - (lastTap[command] || 0) < TAP_LOCK_MS) { return; }
    lastTap[command] = now;

    if (stopped) { return; }
    if (!ws || ws.readyState !== WebSocket.OPEN || state !== 'connected' || !nonce) {
      // Never queue: a command sent late would jump slides unexpectedly.
      showMessage(state === 'waiting'
        ? 'The computer is not connected — that tap was not sent.'
        : 'Not connected — that tap was not sent.', true);
      return;
    }

    var id = nextId++;
    var socket = ws;
    encrypt({ type: command, id: id, nonce: nonce }).then(function (frame) {
      if (ws !== socket || socket.readyState !== WebSocket.OPEN) {
        showMessage('Not connected — that tap was not sent.', true);
        return;
      }
      socket.send(frame);
      ackTimers[id] = setTimeout(function () {
        delete ackTimers[id];
        showMessage('No reply from the computer. Reconnecting… (tap again once connected)', true);
        dropAndReconnect();
      }, ACK_TIMEOUT_MS);
    }).catch(function () {
      showMessage('Not connected — that tap was not sent.', true);
    });

    if (navigator.vibrate) {
      try { navigator.vibrate(12); } catch (e) { /* ignore */ }
    }
  }

  buttons.forEach(function (button) {
    button.addEventListener('click', function () {
      if (button.getAttribute('aria-disabled') === 'true') {
        showMessage(state === 'waiting'
          ? 'The computer is not connected — that tap was not sent.'
          : 'Not connected — that tap was not sent.', true);
        return;
      }
      pressFeedback(button);
      sendCommand(button.getAttribute('data-command'));
      updateWakeLock();
    });
  });

  // Hardware keyboards (and desktop testing): arrows/page keys move slides.
  document.addEventListener('keydown', function (event) {
    if (event.altKey || event.ctrlKey || event.metaKey || event.repeat) { return; }
    var command = null;
    if (event.key === 'ArrowRight' || event.key === 'PageDown') { command = 'presentation.next'; }
    if (event.key === 'ArrowLeft' || event.key === 'PageUp') { command = 'presentation.previous'; }
    if (command) {
      event.preventDefault();
      sendCommand(command);
    }
  });

  // ---------- UI state ----------

  function setState(next) {
    state = next;
    statusEl.setAttribute('data-state', next);
    statusText.textContent = {
      connecting: 'Connecting…',
      waiting: 'Waiting for computer…',
      connected: 'Connected',
      reconnecting: 'Reconnecting…',
      offline: 'Not connected',
      ended: 'Session ended'
    }[next] || next;
    var disabled = next !== 'connected';
    buttons.forEach(function (b) { b.setAttribute('aria-disabled', disabled ? 'true' : 'false'); });
    updateWakeLock();
  }

  function endSession(title, text) {
    stopped = true;
    clearTimeout(reconnectTimer);
    reconnectTimer = null;
    setState('ended');
    showOverlay(title, text, false);
  }

  function pause(title, text) {
    stopped = true;
    setState('offline');
    showOverlay(title, text, true);
  }

  function showOverlay(title, text, canRetry) {
    overlayTitle.textContent = title;
    overlayText.textContent = text;
    overlayRetry.hidden = !canRetry;
    overlay.hidden = false;
    (canRetry ? overlayRetry : overlayTitle).focus();
  }

  overlayRetry.addEventListener('click', function () {
    overlay.hidden = true;
    stopped = false;
    attempt = 0;
    connect();
  });

  function showMessage(text, warning) {
    messageEl.textContent = text;
    messageEl.classList.toggle('is-warning', !!warning);
    clearTimeout(messageTimer);
    messageTimer = setTimeout(clearMessage, MESSAGE_MS);
  }

  function clearMessage() {
    clearTimeout(messageTimer);
    messageEl.textContent = '';
  }

  function pressFeedback(button) {
    button.classList.add('is-pressed');
    setTimeout(function () { button.classList.remove('is-pressed'); }, 120);
  }

  function flash(command, ok) {
    var button = document.querySelector('[data-command="' + command + '"]');
    if (!button) { return; }
    var cls = ok ? 'flash-ok' : 'flash-bad';
    button.classList.add(cls);
    setTimeout(function () { button.classList.remove(cls); }, 300);
  }

  // Keep the phone screen on while presenting (HTTPS lets us use the Wake Lock API).
  // Some browsers (notably iPhone Safari) only grant it during a tap, so every tap retries.
  var wakePending = false;

  function updateWakeLock() {
    var want = state === 'connected' && document.visibilityState === 'visible';
    if (want && !wakeLock && !wakePending && navigator.wakeLock) {
      wakePending = true;
      navigator.wakeLock.request('screen').then(function (lock) {
        wakePending = false;
        wakeLock = lock;
        showAwake(true);
        lock.addEventListener('release', function () {
          wakeLock = null;
          showAwake(false);
        });
      }).catch(function () {
        wakePending = false; // Not allowed right now; the next tap tries again.
        showAwake(false);
      });
    } else if (!want && wakeLock) {
      wakeLock.release().catch(function () { /* ignore */ });
      wakeLock = null;
    }
    if (!want) { showAwake(null); }
  }

  // true: screen is being kept on. false: we tried and could not. null: not presenting.
  function showAwake(on) {
    if (!awakeEl) { return; }
    if (on === null) {
      awakeEl.textContent = '';
    } else if (on) {
      awakeEl.textContent = 'Screen stays on while connected';
    } else if (!navigator.wakeLock) {
      awakeEl.textContent = "Tip: turn off your phone's auto-lock while presenting";
    } else {
      awakeEl.textContent = 'Tap any button to keep the screen on';
    }
  }

  // ---------- lifecycle (lock screen, app switch, network changes) ----------

  function wake() {
    if (stopped || !keys) { return; }
    if (!ws || ws.readyState === WebSocket.CLOSED || ws.readyState === WebSocket.CLOSING) {
      attempt = 0;
      clearTimeout(reconnectTimer);
      reconnectTimer = null;
      connect();
    } else if (ws.readyState === WebSocket.OPEN) {
      // The socket may look open after sleep but be dead. Prove it.
      ping();
    }
    updateWakeLock();
  }

  document.addEventListener('visibilitychange', function () {
    if (document.visibilityState === 'visible') { wake(); } else { updateWakeLock(); }
  });
  window.addEventListener('pageshow', function (event) { if (event.persisted) { wake(); } });
  window.addEventListener('online', wake);
  window.addEventListener('focus', wake);

  // No long-press menus, pinch zoom or double-tap zoom while presenting.
  document.addEventListener('contextmenu', function (e) { e.preventDefault(); });
  document.addEventListener('gesturestart', function (e) { e.preventDefault(); });
  document.addEventListener('dblclick', function (e) { e.preventDefault(); });

  // ---------- helpers ----------

  function getClientId() {
    var key = 'rclicker-client';
    var id = null;
    try { id = window.localStorage.getItem(key); } catch (e) { /* private mode */ }
    if (!id || !/^[A-Za-z0-9_-]{8,64}$/.test(id)) {
      id = toBase64Url(window.crypto.getRandomValues(new Uint8Array(16)));
      try { window.localStorage.setItem(key, id); } catch (e) { /* ignore */ }
    }
    return id;
  }

  // ---------- start ----------

  var keyText = new URLSearchParams(window.location.hash.slice(1)).get('k');
  if (!keyText || !/^[A-Za-z0-9_-]{43}$/.test(keyText)) {
    endSession('Scan the QR code',
      'Open this page by scanning the QR code shown in rclicker on the computer.');
    return;
  }

  if (!window.crypto || !window.crypto.subtle) {
    endSession('Secure page needed',
      'This browser cannot open a secure connection here. Make sure the address starts with https://.');
    return;
  }

  setState('connecting');
  deriveKeys(fromBase64Url(keyText)).then(function (derived) {
    keys = derived;
    connect();
  }).catch(function () {
    endSession('Something went wrong', 'This browser could not set up encryption. Try Safari or Chrome.');
  });
})();
