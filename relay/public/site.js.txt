// Old QR codes pointed at the site root with the session key in the #fragment.
// Send those visitors straight to the phone remote (the key stays in the fragment).
(function () {
  'use strict';
  if (/(^#|&)k=/.test(window.location.hash)) {
    window.location.replace('remote' + window.location.hash);
    return;
  }

  // The address IT should allow is whatever this page was served from.
  document.addEventListener('DOMContentLoaded', function () {
    var host = document.getElementById('site-host');
    if (host) host.textContent = window.location.host;
  });
})();
